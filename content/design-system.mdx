## Code Block

A fenced block with a title bar (copy button included) and highlighted lines:

```tsx title="greet.ts" {6-8}
console.log('hello world');

/**
 * Some comments
 */
function sayHi(name) {
  const message = `hi ${name}`;
  return message;
}
```

A changed line ends in `// [!code ++]` or `// [!code --]`, and copying takes only the new code:

```ts title="greet.ts"
function greet(name: string) {
  return 'Hello ' + name + '!'; // [!code --]
  return `Hello ${name}!`; // [!code ++]
}
```

A word can carry the change too, with `/word/#add` or `/word/#remove` on the fence:

```tsx title="layout.tsx" /suppressHydrationWarning/#add
<html lang="en" suppressHydrationWarning>
```

`variant="danger"` is for errors and failed output:

```txt title="Error" variant="danger"
TypeError: Cannot read properties of undefined (reading 'map')
    at MessageList (components/message-list.tsx:18:22)
    at renderWithHooks (react-dom.development.js:16305:18)
```

A block longer than 24 lines folds to its first 16:

```ts title="easing.ts"
type Easing = (t: number) => number;

export const linear: Easing = (t) => t;

export const easeInQuad: Easing = (t) => t * t;

export const easeOutQuad: Easing = (t) => t * (2 - t);

export const easeInOutQuad: Easing = (t) =>
  t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;

export const easeInCubic: Easing = (t) => t ** 3;

export const easeOutCubic: Easing = (t) => 1 - (1 - t) ** 3;

export const easeInOutCubic: Easing = (t) =>
  t < 0.5 ? 4 * t ** 3 : 1 - (-2 * t + 2) ** 3 / 2;

export const easeOutBack: Easing = (t) => {
  const c1 = 1.70158;
  const c3 = c1 + 1;
  return 1 + c3 * (t - 1) ** 3 + c1 * (t - 1) ** 2;
};

export const easeOutElastic: Easing = (t) => {
  if (t === 0 || t === 1) return t;
  const c4 = (2 * Math.PI) / 3;
  return 2 ** (-10 * t) * Math.sin((t * 10 - 0.75) * c4) + 1;
};

// Runs an easing over a duration, calling back with each eased value.
export function animate(duration: number, ease: Easing, onFrame: (value: number) => void) {
  const start = performance.now();
  const tick = (now: number) => {
    const t = Math.min((now - start) / duration, 1);
    onFrame(ease(t));
    if (t < 1) requestAnimationFrame(tick);
  };
  requestAnimationFrame(tick);
}
```

## Tables

GFM pipes, mapped onto the shadcn primitive.

| Approach | Ships to the browser | Authoring |
| --- | --- | --- |
| GFM pipes | nothing, the plugin runs at build | a row per line |
| `<Table>` composed by hand | nothing, still a Server Component | JSX, for what pipes cannot say |
| TanStack Table | 31 kB, and `'use client'` | a column definition per table |

Alignment comes from the delimiter row.

| Left | Centre | Right |
| :--- | :----: | ----: |
| `--primary` | 262.04 | 1.0 |
| `--border` | 17 presets | 0.5 |
